```js
/**
 * Returns the id of the puzzle immediately before puzzleId in catalogue order.
 * Returns null if puzzleId is first or not found.
 */
export function getPrevId(puzzleId, catalogue = _catalogue) {
  const idx = catalogue.findIndex(e => e.id === puzzleId)
  return idx <= 0 ? null : catalogue[idx - 1].id
}

/**
 * Returns the id of the puzzle immediately after puzzleId in catalogue order.
 * Returns null if puzzleId is last or not found.
 */
export function getNextId(puzzleId, catalogue = _catalogue) {
  const idx = catalogue.findIndex(e => e.id === puzzleId)
  return idx === -1 || idx === catalogue.length - 1 ? null : catalogue[idx + 1].id
}
```

**4. Update `mountGameUi`** to manage puzzle navigation state:

```js
export function mountGameUi(root = document.querySelector('#app')) {
  if (!root) throw new Error('Missing #app mount node')

  // Navigation state
  let currentPuzzleId = null
  let ui = null
  let showingList = false

  function loadPuzzle(puzzleId) {
    ui = createGameUiController({ puzzleId })
    currentPuzzleId = puzzleId
    showingList = false
    rerender()
  }

  function rerender() {
    if (showingList) {
      renderListScreen()
    } else {
      renderGameScreen()
    }
  }

  function renderGameScreen() {
    const model = ui.getRenderModel()
    // extend model with navigation metadata before passing to renderToDom:
    const extModel = {
      ...model,
      puzzleId: currentPuzzleId,
      prevId: getPrevId(currentPuzzleId),
      nextId: getNextId(currentPuzzleId),
    }
    renderToDom(root, extModel)
    // attach nav button listeners after DOM render:
    root.querySelector('[data-prev-puzzle]')?.addEventListener('pointerdown', () => {
      if (extModel.prevId) loadPuzzle(extModel.prevId)
    })
    root.querySelector('[data-next-puzzle]')?.addEventListener('pointerdown', () => {
      if (extModel.nextId) loadPuzzle(extModel.nextId)
    })
    root.querySelector('[data-open-list]')?.addEventListener('pointerdown', () => {
      showingList = true
      rerender()
    })
    root.querySelector('[data-win-banner] [data-next-puzzle]')?.addEventListener('pointerdown', () => {
      if (extModel.nextId) loadPuzzle(extModel.nextId)
    })
  }

  function renderListScreen() {
    const listEl = renderPuzzleList({
      list: ui ? ui.getState().puzzleList ?? [] : [],
      currentId: currentPuzzleId,
      onSelect(id) { loadPuzzle(id) },
      onClose() { showingList = false; rerender() },
    })
    root.innerHTML = ''
    root.append(listEl)
  }

  loadPuzzle(chooseInitialPuzzleId(createController ? createController() : undefined))
  
  root.addEventListener('pointerdown', (event) => {
    if (showingList) return  // list handles its own events
    const cell = event.target.closest?.('[data-cell-key]')
    if (!cell) return
    ui.tapCell(cell.dataset.cellKey)
    rerender()
  })

  return { loadPuzzle, rerender }
}
```

**NOTE:** `renderPuzzleList` needs access to the controller's puzzle list (with locked/solved status). Pass it via `createController().getPuzzleList()` at render time rather than from the UI controller state (since it reflects live localStorage state).
