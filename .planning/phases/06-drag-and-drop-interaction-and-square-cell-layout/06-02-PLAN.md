From src/ui/interactionFeedback.js — createGameUiController.tapCell() semantics:
  - tapCell(key) with no selection → selects piece at key (shows legal moves)
  - tapCell(key) with same key selected → deselects
  - tapCell(destinationKey) with a piece selected → makes the move if legal, or
    selects/illegal-flashes if not legal
  - Used for BOTH selection and move execution; the tap flow is:
      onDragStart → ui.tapCell(fromKey)   // select
      onDrop      → ui.tapCell(toKey)     // move (or re-select/illegal if not legal)
      onCancel    → clear selection explicitly (see below)

From src/ui/interactionFeedback.js — clearSelection():
  feedback.clearSelection() sets selectedKey=null, legalKeys=[]
  Exposed indirectly: tapCell(selectedKey) (same key twice) deselects.
  For cancel, call ui.tapCell(fromKey) a second time to deselect, then call rerender().

Note: createGameUiController does NOT expose feedback.clearSelection() directly.
The cancel path is: if a piece was selected (onDragStart was called), call
ui.tapCell(fromKey) again to deselect it, then rerender(). This is safe because
tapCell(selectedKey === fromKey) → clearSelection path.

──────────────────────────────────────────────────────────────
DRAG THRESHOLD
──────────────────────────────────────────────────────────────
A drag is only recognised if the pointer moves more than DRAG_THRESHOLD (8px) from the
pointerdown origin before pointerup. Below the threshold the interaction is treated as a
tap (onDragStart is NOT called; the module does nothing — the root pointerdown listener
in main.js handles it as a tap).

This is the key mechanism that prevents double-firing: if drag is not triggered, dragDrop
does nothing; the root tap listener fires normally. If drag IS triggered (moved > 8px),
the module calls setPointerCapture and takes over; the tap listener must be suppressed on
the matching pointerup (see main.js Task 2 for how this is wired).

──────────────────────────────────────────────────────────────
MODULE SIGNATURE
──────────────────────────────────────────────────────────────

export function initDragDrop(boardEl, { onDragStart, onDrop, onCancel }) → returns cleanup()

The function attaches pointer event listeners to boardEl and returns a cleanup function
that removes them (needed when the board is torn down and re-rendered).

──────────────────────────────────────────────────────────────
STATE (module-local per initDragDrop call)
──────────────────────────────────────────────────────────────

  let dragState = null
  // When null: no drag in progress
  // When set: {
  //   pointerId,    // for setPointerCapture / releasePointerCapture
  //   fromKey,      // data-cell-key of source cell
  //   fromEl,       // the source cell button element
  //   ghostEl,      // the floating ghost div
  //   startX,       // pointerdown clientX
  //   startY,       // pointerdown clientY
  //   dragging,     // boolean — true once threshold exceeded
  // }

──────────────────────────────────────────────────────────────
EVENT HANDLERS — attach all three to boardEl
──────────────────────────────────────────────────────────────

handlePointerDown(event):
  1. Find the source cell: const cellEl = event.target.closest('[data-cell-key]')
     If no cellEl, return.
  2. Find a piece inside the cell: const pieceEl = cellEl.querySelector('.piece')
     If no pieceEl, return. (Do not start drag tracking on empty cells.)
  3. Store initial drag state (dragging: false) in dragState.
     Set dragState.pointerId = event.pointerId, fromKey, fromEl = cellEl,
     startX = event.clientX, startY = event.clientY.
     Do NOT call setPointerCapture yet (wait until threshold exceeded).
     Do NOT create ghost yet.
     Do NOT call onDragStart yet.

handlePointerMove(event):
  1. If !dragState || event.pointerId !== dragState.pointerId, return.
  2. If !dragState.dragging:
     a. Compute dx = event.clientX - dragState.startX,
              dy = event.clientY - dragState.startY
     b. If Math.hypot(dx, dy) <= DRAG_THRESHOLD (8), return. (Still below threshold.)
     c. Threshold exceeded — activate drag:
        - Set dragState.dragging = true
        - Call boardEl.setPointerCapture(event.pointerId)
        - Create the ghost element (see GHOST CREATION below)
        - Add class 'cell--dragging' to dragState.fromEl
        - Call onDragStart(dragState.fromKey)
  3. If dragState.dragging:
     - Position the ghost: see GHOST POSITIONING below.

handlePointerUp(event):
  1. If !dragState || event.pointerId !== dragState.pointerId, return.
  2. If !dragState.dragging:
     - Clear dragState = null. Return.
     - (Tap path — root listener handles it; we do nothing.)
  3. dragging === true — resolve drop:
     a. Temporarily hide ghost: dragState.ghostEl.style.display = 'none'
     b. const target = document.elementFromPoint(event.clientX, event.clientY)
     c. Show ghost again: dragState.ghostEl.style.display = ''
     d. Find target cell: const targetCellEl = target?.closest('[data-cell-key]')
     e. const toKey = targetCellEl?.dataset.cellKey ?? null
     f. Call cleanupDrag() to remove ghost and dimming.
     g. If toKey && toKey !== dragState.fromKey:
           call onDrop(dragState.fromKey, toKey)
        Else:
           call onCancel(dragState.fromKey)
     h. Set dragState = null.

handlePointerCancel(event):
  1. If !dragState || event.pointerId !== dragState.pointerId, return.
  2. If dragState.dragging: call cleanupDrag(); call onCancel(dragState.fromKey)
  3. Set dragState = null.

Also attach 'pointercancel' to boardEl using handlePointerCancel.

──────────────────────────────────────────────────────────────
GHOST CREATION (called when threshold first exceeded)
──────────────────────────────────────────────────────────────

1. Get the source cell bounding rect: const rect = dragState.fromEl.getBoundingClientRect()
2. Find the piece element: const pieceEl = dragState.fromEl.querySelector('.piece')
3. Clone the piece: const clone = pieceEl.cloneNode(true)
4. Create a wrapper div:
     const ghost = document.createElement('div')
     ghost.className = 'drag-ghost'
5. Set ghost styles directly (these are initial; CSS class also handles them):
     ghost.style.width  = rect.width + 'px'
     ghost.style.height = rect.height + 'px'
     ghost.style.left   = (event.clientX - rect.width / 2) + 'px'
     ghost.style.top    = (event.clientY - rect.height / 2) + 'px'
6. Append clone inside ghost: ghost.append(clone)
7. Append ghost to document.body (NOT the board, so it escapes board overflow clipping)
8. Store as dragState.ghostEl = ghost

──────────────────────────────────────────────────────────────
GHOST POSITIONING (on every pointermove while dragging)
──────────────────────────────────────────────────────────────

Center the ghost on the pointer:
  const rect = dragState.ghostEl.getBoundingClientRect() -- use initial size stored at creation
  dragState.ghostEl.style.left = (event.clientX - dragState.ghostEl.offsetWidth / 2) + 'px'
  dragState.ghostEl.style.top  = (event.clientY - dragState.ghostEl.offsetHeight / 2) + 'px'

  const DRAG_THRESHOLD = 8 // px

  export function initDragDrop(boardEl, { onDragStart, onDrop, onCancel }) {
    let dragState = null

    function cleanupDrag() { ... }

    function handlePointerDown(event) { ... }
    function handlePointerMove(event) { ... }
    function handlePointerUp(event) { ... }
    function handlePointerCancel(event) { ... }

    boardEl.addEventListener('pointerdown',   handlePointerDown)
    boardEl.addEventListener('pointermove',   handlePointerMove)
    boardEl.addEventListener('pointerup',     handlePointerUp)
    boardEl.addEventListener('pointercancel', handlePointerCancel)

    return function cleanup() {
      if (dragState) { cleanupDrag(); dragState = null }
      boardEl.removeEventListener('pointerdown',   handlePointerDown)
      boardEl.removeEventListener('pointermove',   handlePointerMove)
      boardEl.removeEventListener('pointerup',     handlePointerUp)
      boardEl.removeEventListener('pointercancel', handlePointerCancel)
    }
  }
